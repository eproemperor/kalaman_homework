# C++ 智能指针与移动语义完全指南

## 第一部分：移动语义（Move Semantics）

## 1. 左值与右值

### 1.1 基本概念

```cpp
// 左值：有地址、有名字的表达式
int x = 10;      // x 是左值
string s = "hi"; // s 是左值

// 右值：临时值、没有名字的值
int y = 20;      // 20 是右值
int z = x + 5;   // x+5 是右值
string t = s + "world"; // s+"world" 是右值

// 判断方法
void test() {
    int a = 10;
    &a;   // 可以取地址 → 左值
    // &10;  // 错误！不能取临时值的地址 → 右值
}
```

### 1.2 左值引用与右值引用

```cpp
// 左值引用 (Lvalue Reference)
int& lref = a;        // 绑定到左值
const int& clref = 10; // const左值引用可绑定右值

// 右值引用 (Rvalue Reference) - C++11
int&& rref = 10;      // 绑定到右值
rref = 20;            // 可以修改

// 示例
void process(int& x) {
    cout << "Lvalue reference: " << x << endl;
}

void process(int&& x) {
    cout << "Rvalue reference: " << x << endl;
}

int main() {
    int a = 5;
    process(a);   // 调用左值版本
    process(10);  // 调用右值版本
    return 0;
}
```

## 2. 移动构造函数与移动赋值

### 2.1 深拷贝 vs 移动语义

```cpp
class MyBuffer {
private:
    size_t size;
    int* data;
    
public:
    // 构造函数
    MyBuffer(size_t sz = 0) : size(sz), data(sz ? new int[sz] : nullptr) {
        cout << "Constructor" << endl;
    }
    
    // 析构函数
    ~MyBuffer() {
        delete[] data;
        cout << "Destructor" << endl;
    }
    
    // 拷贝构造函数 (深拷贝)
    MyBuffer(const MyBuffer& other) : size(other.size), data(new int[other.size]) {
        cout << "Copy Constructor (Deep Copy)" << endl;
        copy(other.data, other.data + size, data);
    }
    
    // 拷贝赋值运算符
    MyBuffer& operator=(const MyBuffer& other) {
        cout << "Copy Assignment" << endl;
        if (this != &other) {
            delete[] data;
            size = other.size;
            data = new int[size];
            copy(other.data, other.data + size, data);
        }
        return *this;
    }
    
    // 移动构造函数 (转移资源)
    MyBuffer(MyBuffer&& other) noexcept 
        : size(other.size), data(other.data) {
        cout << "Move Constructor (Transfer Resource)" << endl;
        other.size = 0;
        other.data = nullptr;  // 避免double delete
    }
    
    // 移动赋值运算符
    MyBuffer& operator=(MyBuffer&& other) noexcept {
        cout << "Move Assignment" << endl;
        if (this != &other) {
            delete[] data;
            size = other.size;
            data = other.data;
            other.size = 0;
            other.data = nullptr;
        }
        return *this;
    }
};
```

### 2.2 测试移动语义

```cpp
MyBuffer createBuffer() {
    MyBuffer temp(1000);
    return temp;  // 编译器会使用移动语义（NRVO可能优化）
}

int main() {
    MyBuffer b1(100);           // 构造
    MyBuffer b2 = b1;           // 拷贝构造
    MyBuffer b3 = move(b1);     // 移动构造 (b1现在为空)
    
    b2 = b3;                    // 拷贝赋值
    b2 = move(b3);              // 移动赋值
    
    MyBuffer b4 = createBuffer(); // 移动构造或RVO
    
    return 0;
}
```

## 3. std::move 与 std::forward

### 3.1 std::move

```cpp
// std::move 只是将左值转换为右值引用
template<typename T>
decltype(auto) move(T&& param) {
    return static_cast<remove_reference_t<T>&&>(param);
}

// 使用示例
vector<string> v1 = {"hello", "world"};
vector<string> v2 = move(v1);  // 转移资源，v1变为空

cout << "v1 size: " << v1.size() << endl;  // 0
cout << "v2 size: " << v2.size() << endl;  // 2
```

### 3.2 std::forward（完美转发）

```cpp
// 完美转发示例
template<typename T>
void wrapper(T&& arg) {
    // 保持参数的左值/右值属性
    process(forward<T>(arg));
}

void process(int& x) { cout << "Lvalue" << endl; }
void process(int&& x) { cout << "Lvalue" << endl; }

int main() {
    int a = 10;
    wrapper(a);   // 转发左值
    wrapper(20);  // 转发右值
    return 0;
}
```

## 第二部分：智能指针（Smart Pointers）

## 4. unique_ptr（独占所有权）

### 4.1 基本使用

```cpp
#include <memory>

// 创建
unique_ptr<int> p1(new int(10));
unique_ptr<int> p2 = make_unique<int>(20);  // C++14，推荐

unique_ptr<MyClass> p3 = make_unique<MyClass>("Hello");
unique_ptr<int[]> p4 = make_unique<int[]>(100); // 数组

// 所有权转移（不能拷贝，只能移动）
unique_ptr<int> p5 = move(p1);  // p1变为nullptr

// 释放所有权
int* raw = p2.release();  // p2放弃所有权
delete raw;               // 需要手动释放

// 重置
p3.reset();               // 删除对象，置空
p3.reset(new MyClass());  // 删除原对象，接管新对象
```

### 4.2 自定义删除器

```cpp
// 自定义删除器函数
void myDeleter(int* p) {
    cout << "Custom deleting: " << *p << endl;
    delete p;
}

unique_ptr<int, decltype(&myDeleter)> p1(new int(42), myDeleter);

// 用于FILE*等资源
auto fileDeleter = [](FILE* f) {
    if (f) fclose(f);
};
unique_ptr<FILE, decltype(fileDeleter)> filePtr(
    fopen("test.txt", "r"), fileDeleter
);
```

## 5. shared_ptr（共享所有权）

### 5.1 基本使用

```cpp
// 创建
shared_ptr<int> sp1(new int(100));
shared_ptr<int> sp2 = make_shared<int>(200);  // 推荐，更高效

// 引用计数
cout << sp1.use_count() << endl;  // 1

shared_ptr<int> sp3 = sp1;  // 拷贝，引用计数+1
cout << sp1.use_count() << endl;  // 2

sp3.reset();  // 引用计数-1
cout << sp1.use_count() << endl;  // 1

// 获取原始指针
int* raw = sp1.get();
```

### 5.2 循环引用问题

```cpp
// ❌ 错误：循环引用导致内存泄漏
class Node {
public:
    shared_ptr<Node> next;
    ~Node() { cout << "Node destroyed" << endl; }
};

void testCycle() {
    auto node1 = make_shared<Node>();
    auto node2 = make_shared<Node>();
    node1->next = node2;
    node2->next = node1;  // 循环引用！
    // 永远不会输出 "Node destroyed"
}
```

## 6. weak_ptr（弱引用）

### 6.1 基本特性

```cpp
// weak_ptr 不增加引用计数
shared_ptr<int> sp = make_shared<int>(42);
weak_ptr<int> wp = sp;  // 引用计数仍为1

cout << sp.use_count() << endl;  // 1
cout << wp.use_count() << endl;  // 1（相同）

// 检查是否过期
if (wp.expired()) {
    cout << "Resource freed" << endl;
}

// 转换为shared_ptr
if (auto locked = wp.lock()) {
    cout << "Valid: " << *locked << endl;
} else {
    cout << "Resource gone" << endl;
}
```

### 6.2 解决循环引用

```cpp
// ✅ 正确：使用weak_ptr打破循环
class Node {
public:
    int value;
    shared_ptr<Node> next;
    weak_ptr<Node> prev;  // 使用weak_ptr
    
    Node(int v) : value(v) {
        cout << "Node " << v << " created" << endl;
    }
    
    ~Node() {
        cout << "Node " << value << " destroyed" << endl;
    }
};

void testNoCycle() {
    auto node1 = make_shared<Node>(1);
    auto node2 = make_shared<Node>(2);
    
    node1->next = node2;
    node2->prev = node1;  // weak_ptr不会造成循环
    
    // 正常析构
}
```

### 6.3 观察者模式示例

```cpp
class Subject {
private:
    vector<weak_ptr<Observer>> observers;
    
public:
    void attach(shared_ptr<Observer> observer) {
        observers.push_back(observer);
    }
    
    void notify() {
        // 清理已失效的观察者
        observers.erase(
            remove_if(observers.begin(), observers.end(),
                [](const weak_ptr<Observer>& wp) { return wp.expired(); }),
            observers.end()
        );
        
        // 通知有效观察者
        for (auto& wp : observers) {
            if (auto sp = wp.lock()) {
                sp->update();
            }
        }
    }
};
```

## 7. 智能指针对比与选择

### 7.1 性能对比

| 特性 | unique_ptr | shared_ptr | weak_ptr |
|------|------------|------------|----------|
| 内存开销 | 与裸指针相同 | 增加控制块（两个指针） | 与控制块类似 |
| 性能 | 几乎无开销 | 引用计数原子操作开销 | lock()需要原子操作 |
| 拷贝 | 禁止 | 引用计数+1 | 引用计数不变 |
| 移动 | O(1) | O(1) | O(1) |
| 线程安全 | 非线程安全 | 引用计数线程安全 | 引用计数线程安全 |

### 7.2 选择指南

```cpp
// 1. 独占所有权 → unique_ptr
unique_ptr<Resource> res = make_unique<Resource>();

// 2. 共享所有权 → shared_ptr
shared_ptr<Config> globalConfig = make_shared<Config>();

// 3. 观察不拥有 → weak_ptr (打破循环引用)
weak_ptr<Node> parent;

// 4. 简单场景/性能关键 → 裸指针 + 明确所有权
void processData(int* data, size_t size) {
    // 假设调用者保证data有效
}

// 5. 需要数组
unique_ptr<int[]> arr = make_unique<int[]>(100);
// 或使用vector代替
```

## 8. 高级用法与陷阱

### 8.1 enable_shared_from_this

```cpp
// 从this获取shared_ptr
class MyClass : public enable_shared_from_this<MyClass> {
public:
    shared_ptr<MyClass> getShared() {
        return shared_from_this();  // 安全获取
    }
    
    void doSomething() {
        // 需要传递shared_ptr给其他函数
        registerCallback(shared_from_this());
    }
};

// 使用
auto obj = make_shared<MyClass>();
auto sameObj = obj->getShared();  // 正确，共享所有权

// ❌ 错误：使用前对象必须被shared_ptr管理
MyClass obj2;
auto ptr = obj2.getShared();  // 抛出bad_weak_ptr异常
```

### 8.2 自定义删除器高级用法

```cpp
// 管理不同资源
struct Resource {
    int fd;
    void* ptr;
};

void resourceDeleter(Resource* r) {
    if (r->fd != -1) close(r->fd);
    if (r->ptr) free(r->ptr);
    delete r;
}

unique_ptr<Resource, decltype(&resourceDeleter)> 
    res(new Resource{open("file.txt", O_RDONLY), malloc(1024)}, resourceDeleter);
```

### 8.3 别名构造函数

```cpp
struct Data {
    int id;
    string name;
};

// shared_ptr别名构造：共享所有权，指向不同对象
shared_ptr<Data> sp = make_shared<Data>();
shared_ptr<int> alias(sp, &sp->id);  // 指向id，但共享Data的生命周期

// 使用场景：指向成员或子对象
struct Node {
    int value;
    Node* next;
};

shared_ptr<Node> head = make_shared<Node>();
shared_ptr<int> valuePtr(head, &head->value);  // 共享所有权
```

## 9. 实际应用示例

### 9.1 对象池

```cpp
class ObjectPool {
private:
    vector<shared_ptr<Resource>> pool;
    mutex mtx;
    
public:
    shared_ptr<Resource> acquire() {
        lock_guard<mutex> lock(mtx);
        
        for (auto& res : pool) {
            if (res.use_count() == 1) {  // 只有池子持有
                return res;
            }
        }
        
        // 创建新资源
        auto res = make_shared<Resource>();
        pool.push_back(res);
        return res;
    }
};
```

### 9.2 缓存系统

```cpp
template<typename K, typename V>
class Cache {
private:
    unordered_map<K, weak_ptr<V>> cache;
    mutex mtx;
    
public:
    shared_ptr<V> get(const K& key) {
        lock_guard<mutex> lock(mtx);
        
        auto it = cache.find(key);
        if (it != cache.end()) {
            if (auto sp = it->second.lock()) {
                cout << "Cache hit: " << key << endl;
                return sp;
            }
            cache.erase(it);  // 清理失效项
        }
        
        cout << "Cache miss: " << key << endl;
        auto sp = make_shared<V>();
        cache[key] = sp;
        return sp;
    }
};
```

### 9.3 工厂模式结合智能指针

```cpp
template<typename T, typename... Args>
unique_ptr<T> createUnique(Args&&... args) {
    return make_unique<T>(forward<Args>(args)...);
}

template<typename T, typename... Args>
shared_ptr<T> createShared(Args&&... args) {
    return make_shared<T>(forward<Args>(args)...);
}

// 使用
auto uptr = createUnique<MyClass>("name", 100);
auto sptr = createShared<MyClass>("name", 100);
```

## 10. 最佳实践总结

### DO ✅
- 优先使用 `make_unique` / `make_shared`
- 使用 `weak_ptr` 打破循环引用
- 对于工厂函数返回 `unique_ptr`（最少权限原则）
- 使用 `enable_shared_from_this` 获取自身的 `shared_ptr`
- 使用 `std::move` 转移 `unique_ptr` 所有权

### DON'T ❌
- 不要混用裸指针和智能指针管理同一内存
- 不要从 `get()` 返回的裸指针创建新的智能指针
- 不要直接 `delete` 智能指针管理的指针
- 不要在不需要共享所有权时使用 `shared_ptr`
- 不要忽略 `make_shared` 的性能优势（一次分配）

```cpp
// ❌ 错误示例
int* raw = new int(10);
shared_ptr<int> sp1(raw);
shared_ptr<int> sp2(raw);  // 双重删除！

// ✅ 正确
auto sp1 = make_shared<int>(10);
auto sp2 = sp1;

// ❌ 错误
auto sp = make_shared<int>(42);
int* bad = sp.get();
delete bad;  // double free!

// ✅ 正确
sp.reset();  // 由智能指针管理释放
```
