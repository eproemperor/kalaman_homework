# C++ 模板与仿函数完全指南

## 第一部分：模板（Template）

## 1. 函数模板

### 1.1 基本语法

```cpp
#include <iostream>
using namespace std;

// 定义函数模板
template <typename T>
T max(T a, T b) {
    return (a > b) ? a : b;
}

// 使用
int main() {
    cout << max(3, 7) << endl;        // 7 (int)
    cout << max(3.14, 2.78) << endl;  // 3.14 (double)
    cout << max('a', 'z') << endl;    // 'z' (char)
    
    // 显式指定类型
    cout << max<double>(3, 5.2) << endl;
    
    return 0;
}
```

### 1.2 多类型参数

```cpp
template <typename T1, typename T2>
void print(T1 a, T2 b) {
    cout << a << " and " << b << endl;
}

// 返回值类型推导（C++14）
template <typename T1, typename T2>
auto add(T1 a, T2 b) -> decltype(a + b) {
    return a + b;
}

// C++11 trailing return type
template <typename T1, typename T2>
auto multiply(T1 a, T2 b) -> decltype(a * b) {
    return a * b;
}
```

### 1.3 模板重载

```cpp
// 通用版本
template <typename T>
void display(T value) {
    cout << "Generic: " << value << endl;
}

// 特化版本（指针类型）
template <typename T>
void display(T* ptr) {
    cout << "Pointer: " << *ptr << endl;
}

// 普通函数优先于模板
void display(int value) {
    cout << "Normal function: " << value << endl;
}
```

## 2. 类模板

### 2.1 基本定义

```cpp
template <typename T>
class Box {
private:
    T content;
    
public:
    Box(T item) : content(item) {}
    
    void set(T item) {
        content = item;
    }
    
    T get() const {
        return content;
    }
    
    void show() const {
        cout << "Box contains: " << content << endl;
    }
};

// 使用
Box<int> intBox(123);
Box<string> strBox("Hello");
Box<double> doubleBox(3.14);
```

### 2.2 多参数类模板

```cpp
template <typename K, typename V, int CAPACITY = 100>
class SimpleMap {
private:
    pair<K, V> data[CAPACITY];
    int size = 0;
    
public:
    void insert(K key, V value) {
        if (size < CAPACITY) {
            data[size++] = {key, value};
        }
    }
    
    V* find(K key) {
        for (int i = 0; i < size; i++) {
            if (data[i].first == key) {
                return &data[i].second;
            }
        }
        return nullptr;
    }
    
    int getSize() const { return size; }
};

// 使用
SimpleMap<string, int, 50> scores;
scores.insert("Alice", 95);
```

### 2.3 模板特化

```cpp
// 通用模板
template <typename T>
class Comparator {
public:
    bool compare(T a, T b) {
        return a < b;
    }
};

// 完全特化（针对const char*）
template <>
class Comparator<const char*> {
public:
    bool compare(const char* a, const char* b) {
        return strcmp(a, b) < 0;
    }
};

// 偏特化（针对指针）
template <typename T>
class Comparator<T*> {
public:
    bool compare(T* a, T* b) {
        return *a < *b;
    }
};

// 使用
Comparator<int> c1;
cout << c1.compare(3, 5) << endl;  // true

Comparator<const char*> c2;
cout << c2.compare("apple", "banana") << endl;  // true

Comparator<int*> c3;
int x = 10, y = 20;
cout << c3.compare(&x, &y) << endl;  // true
```

### 3.3 SFINAE（Substitution Failure Is Not An Error）

```cpp
#include <type_traits>

// 只接受整数类型
template <typename T>
typename enable_if<is_integral<T>::value, T>::type
half(T value) {
    return value / 2;
}

// 只接受浮点类型
template <typename T>
typename enable_if<is_floating_point<T>::value, T>::type
half(T value) {
    return value / 2.0;
}

// 使用
cout << half(10);     // 调用整数版本
cout << half(3.14);   // 调用浮点版本
// half("hello");     // 编译错误
```

## 第二部分：仿函数（Function Object / Functor）

## 4. 基本概念

仿函数是**重载了operator()的类对象**，行为像函数但有状态。

### 4.1 简单仿函数

```cpp
class Add {
public:
    int operator()(int a, int b) const {
        return a + b;
    }
};

// 使用
Add add;
int result = add(3, 5);  // 像函数一样调用
cout << result << endl;  // 8

// 临时对象调用
cout << Add()(3, 5) << endl;
```

## 5. 有状态仿函数

```cpp
class Counter {
private:
    int count = 0;
    
public:
    int operator()() {
        return ++count;
    }
    
    void reset() {
        count = 0;
    }
    
    int get() const {
        return count;
    }
};

// 使用
Counter c;
cout << c() << endl;  // 1
cout << c() << endl;  // 2
cout << c() << endl;  // 3
c.reset();
cout << c() << endl;  // 1
```

## 6. 仿函数 vs 普通函数

```cpp
// 普通函数
bool greaterThan(int a, int b) {
    return a > b;
}

// 仿函数
class GreaterThan {
private:
    int threshold;
    
public:
    GreaterThan(int t) : threshold(t) {}
    
    bool operator()(int value) const {
        return value > threshold;
    }
};

// 使用
vector<int> nums = {1, 5, 2, 8, 3, 9, 4};

// 普通函数需要固定阈值
int threshold = 5;
auto it1 = find_if(nums.begin(), nums.end(), 
    [threshold](int x) { return x > threshold; });

// 仿函数可以灵活配置
auto it2 = find_if(nums.begin(), nums.end(), GreaterThan(5));
auto it3 = find_if(nums.begin(), nums.end(), GreaterThan(3));
```

## 8. 实用示例

### 8.1 自定义排序仿函数

```cpp
struct Student {
    string name;
    int score;
    int age;
};

// 按分数降序，分数相同按年龄升序
class CompareStudent {
public:
    bool operator()(const Student& a, const Student& b) const {
        if (a.score != b.score) {
            return a.score > b.score;  // 高分在前
        }
        return a.age < b.age;          // 同龄小的在前
    }
};

// 使用
vector<Student> students = {
    {"Alice", 95, 20},
    {"Bob", 85, 19},
    {"Charlie", 95, 18}
};

sort(students.begin(), students.end(), CompareStudent());

for (const auto& s : students) {
    cout << s.name << ": " << s.score << ", " << s.age << endl;
}
// Charlie: 95, 18
// Alice: 95, 20
// Bob: 85, 19
```

### 8.2 数学运算仿函数

```cpp
class Polynomial {
private:
    vector<double> coeffs;  // 系数，从常数项开始
    
public:
    Polynomial(vector<double> c) : coeffs(c) {}
    
    double operator()(double x) const {
        double result = 0;
        double power = 1;
        
        for (double coeff : coeffs) {
            result += coeff * power;
            power *= x;
        }
        return result;
    }
    
    // 导数
    Polynomial derivative() const {
        vector<double> derivCoeffs;
        for (size_t i = 1; i < coeffs.size(); i++) {
            derivCoeffs.push_back(coeffs[i] * i);
        }
        return Polynomial(derivCoeffs);
    }
};

// 使用：f(x) = 3x^2 + 2x + 1
Poly p({1, 2, 3});
cout << p(2) << endl;     // 1 + 4 + 12 = 17

Poly dp = p.derivative();  // f'(x) = 6x + 2
cout << dp(2) << endl;     // 12 + 2 = 14
```

### 8.3 缓存仿函数（Memoization）

```cpp
template <typename T, typename... Args>
class Memoizer {
private:
    map<tuple<Args...>, T> cache;
    function<T(Args...)> func;
    
public:
    Memoizer(function<T(Args...)> f) : func(f) {}
    
    T operator()(Args... args) {
        auto key = make_tuple(args...);
        
        if (cache.find(key) != cache.end()) {
            cout << "Cache hit!" << endl;
            return cache[key];
        }
        
        cout << "Computing..." << endl;
        T result = func(args...);
        cache[key] = result;
        return result;
    }
};

// 使用
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n-1) + fibonacci(n-2);
}

Memoizer<int, int> fibMemo(fibonacci);
cout << fibMemo(5) << endl;  // 计算
cout << fibMemo(5) << endl;  // 缓存命中
```

## 9. 模板 + 仿函数高级应用

### 9.1 函数包装器

```cpp
#include <functional>

// 可以存储任何可调用对象
class TaskManager {
private:
    vector<function<void()>> tasks;
    
public:
    template <typename Callable>
    void addTask(Callable&& task) {
        tasks.emplace_back(forward<Callable>(task));
    }
    
    void executeAll() {
        for (auto& task : tasks) {
            task();
        }
    }
};

// 使用
TaskManager manager;

// 普通函数
void hello() { cout << "Hello "; }
manager.addTask(hello);

// Lambda
manager.addTask([]() { cout << "World "; });

// 仿函数
class PrintNum {
public:
    void operator()() { cout << "123 "; }
};
manager.addTask(PrintNum());

manager.executeAll();  // Hello World 123
```

### 9.2 策略模式配合仿函数

```cpp
template <typename CompressionStrategy>
class DataCompressor {
private:
    CompressionStrategy strategy;
    
public:
    DataCompressor(CompressionStrategy s) : strategy(s) {}
    
    vector<char> compress(const vector<char>& data) {
        return strategy(data);
    }
};

// 不同压缩策略
class ZipStrategy {
public:
    vector<char> operator()(const vector<char>& data) const {
        cout << "Using ZIP compression" << endl;
        // ZIP压缩实现
        return data;
    }
};

class GZipStrategy {
public:
    vector<char> operator()(const vector<char>& data) const {
        cout << "Using GZIP compression" << endl;
        // GZIP压缩实现
        return data;
    }
};

// 使用
DataCompressor<ZipStrategy> zipCompressor(ZipStrategy());
DataCompressor<GZipStrategy> gzipCompressor(GZipStrategy());

vector<char> data = {'H', 'e', 'l', 'l', 'o'};
zipCompressor.compress(data);
gzipCompressor.compress(data);
```

## 11. 最佳实践

### 11.1 优先使用Lambda（简单场景）

```cpp
// 简单排序 → Lambda
sort(v.begin(), v.end(), 
     [](int a, int b) { return a > b; });

// 复杂逻辑 → 仿函数
class ComplexCompare {
public:
    bool operator()(const Data& a, const Data& b) const {
        // 复杂比较逻辑
        if (a.priority != b.priority) return a.priority > b.priority;
        if (a.timestamp != b.timestamp) return a.timestamp < b.timestamp;
        return a.id < b.id;
    }
};
```

### 11.2 模板常见错误

```cpp
// ❌ 错误：模板定义与实现分离（通常情况）
// header.h
template <typename T>
class MyClass {
    void func();
};

// implementation.cpp
template <typename T>
void MyClass<T>::func() {
    // 实现
}

// ✅ 正确：在头文件中实现
// header.h
template <typename T>
class MyClass {
    void func() {
        // 实现
    }
};

// 或者使用显式实例化
// implementation.cpp
template class MyClass<int>;
template class MyClass<double>;
```

## 12. 总结

| 概念 | 用途 | 示例 |
|------|------|------|
| 函数模板 | 通用函数 | `template<typename T> T max(T a, T b)` |
| 类模板 | 通用类 | `template<typename T> class Box { T data; }` |
| 模板特化 | 特定类型定制 | `template<> class Box<bool> {...}` |
| 仿函数 | 有状态函数对象 | `class Add { int operator()(int a, int b) { return a+b; } }` |
| Lambda | 匿名仿函数 | `[](int x){ return x*2; }` |

**选择建议：**
- 需要**通用算法/数据结构** → 模板
- 需要**状态或有多个操作** → 仿函数
- **简单、一次性**使用 → Lambda
- **高性能泛型代码** → 模板 + 仿函数
